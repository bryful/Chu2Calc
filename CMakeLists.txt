cmake_minimum_required(VERSION 3.25)
project(Chu2Calc VERSION 1.1.1 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(chu2core src/Units.cpp src/Motion.cpp src/Energy.cpp src/Orbit.cpp src/BodyPresets.cpp src/Mechanics.cpp src/DragFall.cpp src/Area.cpp)
target_include_directories(chu2core PUBLIC src)
add_executable(Chu2Calc WIN32 src/Main.cpp src/Chu2Calc.manifest src/Chu2Calc.rc)
target_include_directories(Chu2Calc PRIVATE assets)
target_link_libraries(Chu2Calc PRIVATE chu2core comctl32 comdlg32 dwmapi uxtheme)
target_compile_definitions(Chu2Calc PRIVATE UNICODE _UNICODE NOMINMAX)
if(MSVC)
  foreach(target chu2core Chu2Calc)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  endforeach()
  target_link_options(Chu2Calc PRIVATE "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/Chu2Calc.manifest")
endif()
enable_testing()
add_executable(CoreTests tests/CoreTests.cpp)
target_link_libraries(CoreTests PRIVATE chu2core)
if(MSVC)
  target_compile_options(CoreTests PRIVATE /W4 /utf-8)
endif()
add_test(NAME conversions COMMAND CoreTests)
